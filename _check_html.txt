        buf = buf.slice(idx + 2);
        const evLine = raw.split('\n').find(function(l) { return l.startsWith('event:'); });
        const dataLine = raw.split('\n').find(function(l) { return l.startsWith('data:'); });
        if (!dataLine) return;
        const evt = (evLine || '').replace(/^event:\s*/, '').trim() || 'message';
        let data = {};
        try { data = JSON.parse(dataLine.replace(/^data:\s*/, '')); } catch {}
        if (evt === 'token' && data.content) {
          if (typing.parentNode) typing.parentNode.removeChild(typing);
          if (!activeBotBubble) activeBotBubble = renderBubble('bot', '');
          botText += data.content;
          activeBotBubble.innerHTML = md(botText);
          msgs.scrollTop = msgs.scrollHeight;
        } else if (evt === 'tool_call') {
          tools.push({ name: data.name, args: data.arguments });
        } else if (evt === 'tool_result') {
          const last = tools[tools.length - 1];
          if (last) last.result = data.result;
        } else if (evt === 'rag') {
          (data.chunks || []).forEach(function(c) { ragSources.push(c); });
        } else if (evt === 'done') {
          if (typing.parentNode) typing.parentNode.removeChild(typing);
          if (!activeBotBubble) activeBotBubble = renderBubble('bot', md(botText));
          finalBubble = activeBotBubble;
        } else if (evt === 'error') {
          if (typing.parentNode) typing.parentNode.removeChild(typing);
          renderBubble('error', '⚠️ ' + (data.message || 'unknown error'));
          finalBubble = null;
        }
      }
      while (true) {
        const { value, done } = await reader.read();
        if (done) {
          // Stream exhausted — process any trailing data still in buf (no \n\n needed)
          buf = buf.trim();
          if (buf) processBuf();
          break;
        }
        buf += dec.decode(value, { stream: true });
        // Drain all complete events in buf
        while (buf.indexOf('\n\n') !== -1) processBuf();
      }
    } catch (err) {
      if (typing.parentNode) typing.parentNode.removeChild(typing);
      const isAbort = err && (err.name === 'AbortError' || (err.message && err.message.includes('abort')));
      renderBubble('error', isAbort
        ? '⚠️ 連線逾時（後端服務喚醒中，請稍候後再試）'
        : '⚠️ 連線失敗：' + (err && err.message ? err.message : String(err)));
      finalBubble = null;
    } finally {
      sendBtn.disabled = false;
      activeBotBubble = null;
    }

    // append footnotes to the final bot bubble (runs after finally so we have a stable ref)
    if (finalBubble) {
      if (tools.length) {
        const div = document.createElement('div');
        div.className = 'cw-tools';
        div.innerHTML = tools.map(function(t) { return '<span class="cw-tool-pill">🔧 ' + escapeHtml(t.name) + '</span>'; }).join('');
        finalBubble.appendChild(div);
      }
      if (ragSources.length) {
        const div = document.createElement('div');
        div.className = 'cw-rag-sources';
        const uniq = [];
        const seen = new Set();
        ragSources.forEach(function(s) {
          const key = s.url || s.title;
          if (!seen.has(key)) { seen.add(key); uniq.push(s); }
        });
        div.innerHTML = '📚 引用：' + uniq.slice(0, 5).map(function(s) {
          const url = s.url || '#';
          const title = s.title || url;
          return '<a href="' + escapeHtml(url) + '" target="_blank" rel="noopener">' + escapeHtml(title) + '</a>';
        }).join(' · ');
        finalBubble.appendChild(div);
      }
    }
    if (botText) history.push({ role: 'assistant', content: botText });
    saveHistory(history);
  }
})();
